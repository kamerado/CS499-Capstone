// sort vector.
Fucntion partition(vector<Course>& courses, int lowIdx, int highIdx) {
  // pick middle element as pivot
  midpoint = lowIdx + (highIdx - lowIdx) / 2
  pivot = courses.at(midpoint).courseNumber

  done = false
  while (!done) {
    // increment lowIdx while nums[lowIdx] less than pivot
    while (courses.at(lowIdx).courseNumber less than pivot) {
      lowIdx++
    }

    // decriment highIdx while pivot < nums[highIdx]
    while (pivot less than courses.at(highIdx).courseNumber) {
      highIdx--
    }

    // if zero or one element remain, all numbers are
    // partitioned. return highIdx
    if (lowIdx greater than or equal to highIdx) {
      done = true
    } else {
      // swap nums[lowIdx] and nums[highIdx]
      Course tmp = courses.at(lowIdx)
      courses.at(lowIdx) = courses.at(highIdx)
      courses.at(highIdx) = tmp

      // update lowIdx and highIdx
      lowIdx++
      highIdx--
    }
  }

  return highIdx
}

// quicksort to sort alphanumerically
Function quickSort(vector<Course>& courses, int lowIdx, int highIdx) {
  // base case: if partition size is 1 or 0,
  // the partition is already sorted.
  if (lowIdx >= highIdx) {
    return;
  }

  // partition the data within the array. Value lowEndIdx
  // returned from partitioning is the index of the low
  // partitions last index.
  int lowEndIdx = partition(courses, lowIdx, highIdx);

  // Recursivly sort low partition (lowIdx to lowEnxIdx)
  // and high partition (lowEnxIdx to highIdx)
  quickSort(courses, lowIdx, lowEndIdx);
  quickSort(courses, lowEndIdx + 1, highIdx);
}
